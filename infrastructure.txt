# infrastructure.txt
# External Infrastructure Dependencies for Nova
# These are NOT Python packages - they need to be installed separately!

# ============================================================
# DATABASES
# ============================================================

# PostgreSQL (Recommended for production)
# - Primary relational database
# - Version: 13.x, 14.x, 15.x
# - Install: https://www.postgresql.org/download/
# - Docker: postgres:15-alpine
# - Port: 5432
# - Nova connection: postgresql://user:pass@host:5432/dbname
postgresql>=13.0

# MySQL (Alternative relational DB)
# - Version: 8.x
# - Install: https://dev.mysql.com/downloads/
# - Docker: mysql:8
# - Port: 3306
# - Nova connection: mysql://user:pass@host:3306/dbname
mysql>=8.0

# MongoDB (Document database)
# - Version: 5.x, 6.x
# - Install: https://www.mongodb.com/try/download/community
# - Docker: mongo:6
# - Port: 27017
# - Nova connection: mongodb://host:27017
mongodb>=5.0

# ============================================================
# CACHING
# ============================================================

# Redis (Primary cache)
# - Version: 7.x
# - Install: https://redis.io/download
# - Docker: redis:7-alpine
# - Port: 6379
# - Nova connection: redis://host:6379
redis>=7.0

# Memcached (Alternative cache)
# - Version: 1.6.x
# - Install: https://memcached.org/downloads
# - Docker: memcached:1.6
# - Port: 11211
# - Nova connection: memcached://host:11211
memcached>=1.6.0

# ============================================================
# MESSAGE QUEUES & STREAMING
# ============================================================

# Apache Kafka (Event streaming)
# - Version: 3.x
# - Install: https://kafka.apache.org/downloads
# - Docker: confluentinc/cp-kafka:latest
# - Port: 9092 (broker), 2181 (zookeeper)
# - Nova connection: kafka://host:9092
kafka>=3.0

# RabbitMQ (Message broker)
# - Version: 3.10.x
# - Install: https://www.rabbitmq.com/download.html
# - Docker: rabbitmq:3-management
# - Port: 5672 (AMQP), 15672 (management)
# - Nova connection: amqp://host:5672
rabbitmq>=3.10
